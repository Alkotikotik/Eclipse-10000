//Fusion-38 ISA
[31:n] opcode
[n:k]  operands
[k:0]  permutations / empty

Every instruction is 32 bits. [31:29] is the instruction group, for encoding primaraly
000 : FMA
001 : IFMA
010 : XMU (XMU-T and XMU-M)
011 : Memory
100 : GUCH (everything that isn't FMA or IFMA)
101 : reserved (branches / masks)
110 : reserved (PACK control)
111 : reserved (uniforms / XMU-T slot config / future units)

//Registers
Register fields are 6 bits. rw0 - rw47 are actual registers, codes 48 - 63 are constants.
Each lane has 3 banks (1 BRAM18 each), bank = reg[5:4], BRAM address = {pack_id, reg[3:0]}.
Register fields sit in the same place in every group: rd [28:23], ra [22:17], rb [16:11], rc [10:5].
A register field an instruction doesn't use holds parameters instead.

float     |     int
48 : 1.0        1
49 : -1.0       -1 (all ones)
50 : 0.5        2
51 : 2.0        4
52 : 0.25       8
53 : 4.0        12
54 : 1/(2pi)    255
55 : log2(e)    0xFFFF
56 : ln2        31
57 : U0         //runtime programmable registers
58 : U1
59 : U2
60 : U3
61 : GUCH_ID    //0 - 11
62 : PACK_ID    //0 - 31
63 : 0          0           //discard the result

//Compare codes
00 : ==
01 : !=
10 : <
11 : <=

//Round codes (int results)
00 : trunc
01 : floor
10 : ceil
11 : round (ties to even)

//Memory
32MB, 32-bit words only, 23-bit word address. Pixels are RGB565, 2 per word.

FMA <<==
[31:29] 000
[28:23] rd
[22:17] ra
[16:11] rb
[10:5]  rc
[4]     neg_p
[3]     neg_c
[2]     abs_a
[1]     abs_efp
[0]     sat_01
==>>

FMA rw0 <<== +-[|rw1| * rw2] +- rw3 <| <ABS>, <SAT01>
ADD and MUL are FMA with rb = 1.0 or rc = 0. ldexp is ra = 0 (see FCMP for c * 2^m).

IFMA <<==
[31:29] 001
[28:24] k      (0 - 16)
[23]    unused
[22:17] ra
[16:11] rb
[10:5]  rc     (source and destination)
[4]     signed
[3]     neg_c  (negates the whole result)
[2]     neg_p  (rb is unsigned)
[1:0]   unused
==>>
IFMA rc, ra, rb, k      //rc = +-((ra * rb + (rc << k)) >> k), >> k is floor

XMU <<==
[31:29] 010
[28:23] rd
[22:17] ra
[16:11] rb
[10:5]  rc
[4:1]   trans_op / misc_op //XMU-op btw
[0]     efp
==>>

XMU-T (trans_op = XMU-T opcode, only rd and ra are used):
0000 : RSQ   rd, ra
0001 : RCP   rd, ra
0010 : LOG2  rd, ra
0011 : EXP2  rd, ra
0100 : SIN   rd, ra    //input in turns
1100 : COS   rd, ra    //input in turns
0101 : ATAN  rd, ra    //output in turns, |ra| >= 1 takes 2 passes
0110 : SQRT  rd, ra
0111 : SLOT  rd, ra    //programmable function, config is static (group 111)
1110 : FRACT rd, ra

XMU-M (misc_op = [3:1] = XMU-M opcode):
1000 : MIN    rd, ra, rb          //efp = float, otherwise rc[0] = signed
1001 : MAX    rd, ra, rb          //efp = float, otherwise rc[0] = signed, SAM feeds ra/rb into XMU-M b/c
1111 : CLAMP  rd, ra, rb, rc      //rb clamped between ra and rc, efp = float, otherwise signed
1010 : POPCNT rd, ra
1011 : BITREV rd, ra
1101 : reserved

Memory <<==
[31:29] 011
[28:23] rd / rs
[22:17] ra     (base)
[16:3]  imm14  (signed word offset)
[2:0]   mem_op
==>>

000 : LOAD  rd, [ra + imm14]
001 : STORE rs, [ra + imm14]
010 - 111 : reserved

GUCH <<==
[31:29] 100
[28:5]  depends on main_op
[4:2]   main_op
[1:0]   depends on main_op
==>>

FCMP (main_op 000) <<==
[28:25] m      (signed, -8 to 7)
[24:23] cmp
[22:17] ra
[16:11] rb
[10:5]  rc
[4:2]   000
[1]     neg_p
[0]     neg_c
==>>

FCMP ra, rb, rc, m      //cmp_tru = +-ra * rb <cmp> +-rc * 2^m, > and >= by flipping neg_p and neg_c

IMMAD (main_op 001) <<==
[28:23] rd
[22:17] ra
[16:11] rb
[10:5]  rc
[4:2]   001
[1]     signed
[0]     neg_c  (negates the whole result)
==>>

IMMAD rd, ra, rb, rc    //rd = +-(ra * rb + rc), global id = IMMAD rd, PACK_ID, 12, LANE_ID

LOMUL (main_op 010) <<==
[28:23] rd
[22:17] ra
[16:11] rb
[10:5]  k      (0 - 16)
[4:2]   010
[1:0]   unused
==>>

LOMUL rd, ra, rb, k     //rd = (ra * rb) >> k

HIMUL (main_op 011) <<==
[28:23] rd
[22:17] ra
[16:11] rb
[10:5]  rc
[4:2]   011
[1:0]   unused
==>>

HIMUL rd, ra, rb, rc    //rd = (ra[31:24] * rb + rc) >> 8, rc is PREMUL's result

ALU (main_op 100) <<==
[28:23] rd
[22:17] ra
[16:11] rb     (or k for immediate shifts)
[10:5]  alu_op
[4:2]   100
[1:0]   unused
==>>

000000 : ADD    rd, ra, rb
000001 : SUB    rd, ra, rb
000010 : AND    rd, ra, rb
000011 : OR     rd, ra, rb
000100 : XOR    rd, ra, rb
000101 : PREMUL rd, ra, rb     //rd = (ra[23:0] * rb) >> 24, unsigned
001000 : SHL    rd, ra, rb     //shift by rb[4:0]
001001 : SHR    rd, ra, rb
001010 : SRA    rd, ra, rb
001100 : SHLI   rd, ra, k      //k = 0 - 31
001101 : SHRI   rd, ra, k
001110 : SRAI   rd, ra, k      //k = 0 - 29
010000 : I2F    rd, ra
0101cc : F2I    rd, ra         //cc = round code
100scc : ICMP   ra, rb         //pred = ra <cc> rb, s = signed

main_op 101 - 111 : reserved
